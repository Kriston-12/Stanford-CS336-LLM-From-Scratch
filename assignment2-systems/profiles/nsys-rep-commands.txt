# medium_seq128 (corrected, with --cuda_profiler_api added)
sudo "$(command -v nsys)" profile \
  -o ../profiles/medium_seq128 --capture-range=cudaProfilerApi --capture-range-end=stop --force-overwrite=true \
  --trace=cuda,cudnn,cublas,osrt,nvtx \
  --pytorch=functions-trace,autograd-shapes-nvtx \
  --cudabacktrace=all \
  --python-backtrace=cuda \
  --gpu-metrics-devices=0 \
  "$(command -v python)" benchmarking_script.py \
  --model_size medium --seq_len 128 --batch_size 16 --warmup_steps 2 --cuda_profiler_api --annotated_attention
